function parseCSV(txt) {
const rows = [];
for (let line of txt.trim().split('\n')) {
const cells = []; let cur = '', inQ = false;
for (let ch of line) {
if (ch === '"') { inQ = !inQ; }
else if (ch === ',' && !inQ) { cells.push(cur.trim()); cur = ''; }
else cur += ch;
}
cells.push(cur.trim());
rows.push(cells);
}
return rows;
}

function num(s) { return parseFloat(String(s||'').replace(/[$,\s]/g,'')) || 0; }

function peso(n, full) {
const abs = Math.abs(n), sign = n < 0 ? '−' : '';
if (full) return sign + '$' + Math.round(abs).toLocaleString('es-AR');
if (abs >= 1000000) return sign + '$' + (abs/1000000).toFixed(1) + 'M';
if (abs >= 1000) return sign + '$' + Math.round(abs/1000) + 'K';
return sign + '$' + Math.round(abs).toLocaleString('es-AR');
}
